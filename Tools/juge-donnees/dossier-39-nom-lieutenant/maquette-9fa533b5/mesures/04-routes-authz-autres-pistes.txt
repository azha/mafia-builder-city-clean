  /** Resolve account_id → player_id via the 1-1 Player↔Account link (the GET /v1/me identity
   *  bridge — duplicated verbatim, the established per-controller convention, never shared). */
  private async resolvePlayerId(accountId: string): Promise<string> {
    const rows = await this.db
      .select({ player_id: player.player_id })
      .from(player)
      .innerJoin(account, eq(account.account_id, player.account_id))
      .where(and(eq(player.account_id, accountId), eq(account.kind, 'PLAYER')))
      .limit(1);
    const playerId = rows[0]?.player_id;
    if (!playerId) {
      throw new ApiError('RESOURCE_NOT_FOUND', { message: 'No player profile for this account.' });
    }
    return playerId;
  }

export function nomLieutenantTrainDeVie(
  rows: readonly LifestyleRowJointe[],
  playerId: string,
  bucket: LifestyleAlarmBucket,
): string | null {
  if (bucket === 'quiet' || rows.length === 0) return null;
  const { worstGap, worstConsecutive, worstStage } = maximaTrainDeVie(rows);
  const auMaximum = rows.filter((row) => {
    const { gap, consecutive, stage } = normalise(row);
    return stage === worstStage && consecutive === worstConsecutive && gap === worstGap;
  });
  if (auMaximum.length !== 1) return null;
  const [seule] = auMaximum;
  // M2 : aucune FK composite ne lie `lifestyle_audit_state.player_id` à `lieutenant.player_id` — la
  // jointure SQL est scopée, et on le revérifie ici (le nom d'un lieutenant d'un autre joueur ne sort jamais).
  if (seule.lieutenant_player_id !== playerId) return null;
  if (seule.lieutenant_name === null || seule.lieutenant_name === PLACEHOLDER_NOM_LIEUTENANT) return null;
  return seule.lieutenant_name;
}

export function maximaTrainDeVie(rows: readonly LifestyleRowJointe[]): MaximaTrainDeVie {
  let worstGap: number | null = null;
  let worstConsecutive = 0;
  let worstStage: LifestyleStage = 'none';
  for (const row of rows) {
    const { gap, consecutive, stage } = normalise(row);
    if (LIFESTYLE_STAGE_ORDER[stage] > LIFESTYLE_STAGE_ORDER[worstStage]) worstStage = stage;
    if (consecutive > worstConsecutive) worstConsecutive = consecutive;
    if (gap !== null && (worstGap === null || gap > worstGap)) worstGap = gap;
  }
  return { worstGap, worstConsecutive, worstStage };
}
