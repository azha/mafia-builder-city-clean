## 3. Le run — les 13 catégories, UN lancement

```bash
env -u MAFIA_CAPTURE_IDENTIFIER -u MAFIA_CAPTURE_PASSWORD -u MAFIA_DEMO_IDENTIFIER -u MAFIA_DEMO_PASSWORD \
  LOG_FILE="$S/creneau.log" TIMEOUT_S=3600 WT="$PWD" \
  MAFIA_CI_CATEGORIES="EcranAppro,EcranDistribution,EcranDemolition,I18n,BundleReel,W3U1,EcranAutonomy,EcranExceptions,Ecran10,EcranUiLieutenant,ScreenCarte,DemoIdentity,EcranRegleLieutenant" \
  bash Tools/run-unity-check.sh -executeMethod MafiaCI.RunPlayModeTests > "$S/creneau.stdout" 2>&1; echo "RC_RUN=$?"
grep -aoE 'elapsed=[0-9]+s' "$S/creneau.stdout"
grep -aoE 'DÉCLARÉS SOUS LE FILTRE *: *[0-9]+' "$S/creneau.log"
grep -aoE 'RunPlayModeTests finished — passed=[0-9]+ failed=[0-9]+[^"]*' "$S/creneau.log"
grep -a 'catégories RÉELLEMENT exécutées' "$S/creneau.log"
grep -aE 'SUITE TRONQUÉE|FILTRE VIDE|filtre SURCHARGÉ' "$S/creneau.log"
grep -a 'MafiaCI: FAIL ' "$S/creneau.log"
grep -acE 'SIGSEGV|Segmentation fault|core dumped' "$S/creneau.log"
```
